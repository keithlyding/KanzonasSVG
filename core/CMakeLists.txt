# KanzonasSVG engine: one shared library (kzsvg.dll / libkzsvg.so) with no outside
# dependencies: Inkscape's trace code, Potrace 1.16 and Autotrace 0.40, all built here.
cmake_minimum_required(VERSION 3.16)
project(kzsvg LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 99)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Potrace 1.16, the library part (see third_party/potrace/README.md)
set(PT third_party/potrace)
add_library(potrace STATIC ${PT}/curve.c ${PT}/trace.c ${PT}/decompose.c ${PT}/potracelib.c)
target_include_directories(potrace PUBLIC ${PT})
target_compile_definitions(potrace PRIVATE VERSION="1.16" HAVE_INTTYPES_H)

# Autotrace 0.40, as bundled with Inkscape, without its input/output modules and GLib
set(AT third_party/autotrace)
file(GLOB AT_SOURCES ${AT}/src/autotrace/*.c)
add_library(autotrace STATIC ${AT_SOURCES})
target_include_directories(autotrace PUBLIC ${AT}/shim ${AT}/include PRIVATE ${AT}/include/autotrace ${AT}/src/autotrace)
target_compile_definitions(autotrace PRIVATE HAVE_CONFIG_H _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  # M_PI (vector.c asks for it only after math.h is already in); conversion warnings of old C code
  target_compile_definitions(autotrace PRIVATE _USE_MATH_DEFINES)
  target_compile_options(autotrace PRIVATE /wd4005 /wd4244 /wd4267 /wd4305 /wd4996)
  target_compile_options(potrace PRIVATE /wd4244 /wd4267 /wd4996)
else()
  target_compile_options(autotrace PRIVATE -Wno-int-in-bool-context -Wno-address)
endif()

add_library(kzsvg SHARED src/engine.cpp src/drawing.cpp src/filterset.cpp src/imagemap.cpp src/quantize.cpp)
target_link_libraries(kzsvg PRIVATE potrace autotrace)
if(UNIX)
  target_link_libraries(kzsvg PRIVATE m)
endif()
if(MSVC)
  target_compile_definitions(kzsvg PRIVATE _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
  # one DLL, no Visual C++ runtime to install
  set_property(TARGET kzsvg potrace autotrace PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
elseif(WIN32)
  target_link_options(kzsvg PRIVATE -static -static-libgcc -static-libstdc++)
else()
  target_link_options(kzsvg PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL)
endif()
# kzsvg.dll on Windows (not libkzsvg.dll with MinGW)
if(WIN32)
  set_target_properties(kzsvg PROPERTIES PREFIX "")
endif()
